function toggleAuto(type) { if (type === 'forex') { autoF = !autoF; document.getElementById('toggleForex').classList.toggle('on', autoF); // Отправка в бота if (window.Telegram && window.Telegram.WebApp) { window.Telegram.WebApp.sendData(JSON.stringify({auto_forex: autoF})); } showToast('Forex авто: ' + (autoF ? '🟢 ВКЛ' : '🔴 ВЫКЛ')); } else { autoO = !autoO; document.getElementById('toggleOtc').classList.toggle('on', autoO); if (window.Telegram && window.Telegram.WebApp) { window.Telegram.WebApp.sendData(JSON.stringify({auto_otc: autoO})); } showToast('OTC авто: ' + (autoO ? '🟢 ВКЛ' : '🔴 ВЫКЛ')); } } function requestSignal(market) { if (window.Telegram && window.Telegram.WebApp) { window.Telegram.WebApp.sendData(JSON.stringify({request_signal: market})); window.Telegram.WebApp.close(); } showToast('✅ Запрос отправлен'); }
